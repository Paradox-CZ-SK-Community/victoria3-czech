##################################################
# CZECH GRAMMAR - MACROS
##################################################
#
# Short forms of declined names (customizable localization from
# common/customizable_localization/99_pl_*.txt). XCASE is a quoted
# case letter: 'N', 'G', 'D', 'A', 'L', 'I'.
#
#   [CZParty(XPARTY,'G')]    = [P.GetCustom('PARTY_G_TT')]  (tooltip like GetName)
#   [CZPartyNF(XPARTY,'G')]  = [P.GetCustom('PARTY_G')]     (plain text)
#
# Same for CZGoods (GDS_*), CZIG (IG_*) and CZMarket (MARKET_*).
##################################################

macro = {
	description = "Declined party name with tooltip, cases N G D A L I"
	definition = "CZParty(XPARTY,XCASE)"
	replace_with = "XPARTY.GetCustom(Concatenate('PARTY_',Concatenate(XCASE,'_TT')))"
}

macro = {
	description = "Declined party name without formatting, cases N G D A L I"
	definition = "CZPartyNF(XPARTY,XCASE)"
	replace_with = "XPARTY.GetCustom(Concatenate('PARTY_',XCASE))"
}
