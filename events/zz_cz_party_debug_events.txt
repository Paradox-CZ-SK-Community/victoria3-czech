	type = country_event
	placement = root
	title = cz_party_debug.1.t
	desc = cz_party_debug.1.d
	flavor = cz_party_debug.f

	event_image = {
		video = "southamerica_election"
	}

	duration = 3

	immediate = {
		py:agrarian_party ?= { save_scope_as = cz_dbg_agrarian_party }

		py:military_party ?= { save_scope_as = cz_dbg_military_party }
		py:radical_party ?= { save_scope_as = cz_dbg_radical_party }
		py:religious_party ?= { save_scope_as = cz_dbg_religious_party }
		py:revolutionary_party ?= { save_scope_as = cz_dbg_revolutionary_party }
		py:social_democrat_party ?= { save_scope_as = cz_dbg_social_democrat_party }
		random_interest_group = {
			limit = { has_party = yes }
			party = { save_scope_as = cz_dbg_party_test }
		}
	}

	option = {
		name = cz_party_debug.next
		trigger_event = { id = cz_party_debug.2 days = 0 popup = yes }
	}

	option = {
		name = cz_party_debug.close
		default_option = yes
	}
}
